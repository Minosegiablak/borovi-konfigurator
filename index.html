function fv(id){var e=document.getElementById(id);return e?parseFloat(e.value)||0:0;}
function iv(id){var e=document.getElementById(id);return e?parseInt(e.value)||0:0;}

function calcPrice(){
var sk=cfg.surface==='dekor'?'d':cfg.surface==='cpl'?'c':'m';
var listaAr=cfg.model.p[sk]||0;
var mult=cfg.type==='double'?(cfg.model.id==='bizanc'?2:1.8):1;
var listaSz=Math.round(listaAr*mult);
var kedv=fv('kedv-pct')/100;
var hasMod=fv('has-mod')/100;
var hasFel=fv('has-fel')/100;
function feToElad(at){return Math.round(at*(1+hasFel));}
var nettoModell=Math.round(listaSz*(1-kedv));
var eladModell=Math.round(nettoModell*(1+hasMod));
if(!listaAr){nettoModell=0;eladModell=0;}
if(cfg.frame==='none'&&listaAr){
var lapAt=sk==='d'?iv('f_lapd'):sk==='c'?iv('f_lapc'):iv('f_lapm');
var lapSz=cfg.type==='double'?lapAt*2:lapAt;
eladModell=Math.round(Math.round(lapSz*(1-kedv))*(1+hasMod));
}
var items=[];
if(eladModell>0)items.push({label:cfg.model.name+(cfg.frame==='none'?' (csak lap)':''),val:eladModell});
// gerébtok
if(cfg.frame==='gereb'){
var gAt=cfg.surface==='mdf'?iv('f_gmdf'):iv('f_gdc');
items.push({label:'Gerébtok felár',val:feToElad(cfg.type==='double'?gAt*2:gAt)});
}
// tokszélesítés
var falv=cfg.falv||10;
var isD=cfg.surface==='dekor',lim=isD?12:15;
if(falv>lim&&cfg.frame!=='none'){
var extra=falv-lim;
var tokFelAr=Math.round(extra*(isD?iv('f_tokd'):iv('f_tokc')));
if(tokFelAr>0)items.push({label:'Tokszélesítés (+'+extra.toFixed(1)+' cm)',val:feToElad(tokFelAr)});
}
// ral
if(cfg.surface==='mdf'&&cfg.ralCode.trim())items.push({label:'RAL szín felár',val:feToElad(iv('f_ral'))});
// küszöb
if(cfg.kuszob==='cpl'){
var tw=falv;
var cAt=tw<=13?iv('f_c1'):tw<=20?iv('f_c2'):tw<=26?iv('f_c3'):tw<=33?iv('f_c4'):tw<=40?iv('f_c5'):tw<=50?iv('f_c6'):iv('f_c7');
items.push({label:'CPL küszöb',val:feToElad(cAt)});
}
if(cfg.kuszob==='auto')items.push({label:'Automata küszöb',val:feToElad(iv('f_auto'))});
if(cfg.zar!=='bb')items.push({label:'WC/PZ zár felár',val:feToElad(iv('f_zar'))});
if(cfg.szello==='fem')items.push({label:'Fém szellőzőrács',val:feToElad(iv('f_sz'))});
if(cfg.betet==='furat')items.push({label:'Furatolt betét',val:feToElad(iv('f_fur'))});
if(cfg.betet==='lenrost')items.push({label:'Lenrost betét',val:feToElad(iv('f_len'))});
if(cfg.csuko==='dorma')items.push({label:'Dorma ajtócsukó',val:feToElad(iv('f_dor'))});
return{items:items,total:items.reduce(function(s,r){return s+r.val;},0),listaAr:listaAr};
}
